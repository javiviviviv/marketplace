package com.polec.mscarrito.controller;

import com.polec.mscarrito.client.ItemClient;
import com.polec.mscarrito.client.ItemDTO;
import com.polec.mscarrito.dto.AgregarItemRequest;
import com.polec.mscarrito.entity.Carrito;
import com.polec.mscarrito.entity.CarritoItem;
import com.polec.mscarrito.repository.CarritoRepository;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;

import java.util.List;
import java.util.Optional;

@RestController
@RequestMapping("/carritos")
public class CarritoController {
    private final CarritoRepository repository;
    private final ItemClient itemClient;

    public CarritoController(CarritoRepository repository, ItemClient itemClient) {
        this.repository = repository;
        this.itemClient = itemClient;
    }

    @PostMapping("/items")
    public ResponseEntity<Carrito> agregarItem(@RequestBody AgregarItemRequest request) {

        ItemDTO itemOriginal = itemClient.obtenerItemPorId(request.itemId());

        Optional<Carrito> carritoExistente = repository.findByUsuarioId(request.usuarioId());
        Carrito carrito;

        if (carritoExistente.isPresent()) {
            carrito = carritoExistente.get();
        } else {
            carrito = new Carrito();
            carrito.setUsuarioId(request.usuarioId());
        }

        CarritoItem nuevoItem = new CarritoItem();
        nuevoItem.setItemId(itemOriginal.id());

        nuevoItem.setItemPrecio(itemOriginal.precio());

        nuevoItem.setCarrito(carrito);
        carrito.getItems().add(nuevoItem);

        Carrito carritoGuardado = repository.save(carrito);

        return ResponseEntity.status(HttpStatus.CREATED).body(carritoGuardado);
    }
    @GetMapping
    public List<Carrito> listarCarritos(){
        return repository.findAll();
    }


    @DeleteMapping("/{usuarioId}")
    public ResponseEntity<Void> eliminar(@PathVariable Long usuarioId){
        if(!repository.existsByUsuarioId(usuarioId)){
            return ResponseEntity.notFound().build();
        }
        repository.deleteByUsuarioId(usuarioId);
        return ResponseEntity.noContent().build();
    }

    @GetMapping("/{usuarioId}")
    public ResponseEntity<Carrito> verCarrito(@PathVariable Long usuarioId) {
        return repository.findByUsuarioId(usuarioId)
                .map(ResponseEntity::ok)
                .orElse(ResponseEntity.notFound().build());
    }



}

package com.polec.msitem.controller;

import com.polec.msitem.client.CategoriaClient;
import com.polec.msitem.client.CategoriaDTO;
import com.polec.msitem.dto.ItemResponse;
import com.polec.msitem.entity.Item;
import com.polec.msitem.repository.ItemRepository;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;

import java.util.List;

@RestController
@RequestMapping("/items")
public class ItemController {
    //inyectar el repository
    private final ItemRepository repository;
    //inyectar el objeto rest
    private final CategoriaClient categoriaClient;

    public ItemController(ItemRepository repository, CategoriaClient categoriaClient){
        this.repository = repository;
        this.categoriaClient = categoriaClient;
    }

    @PostMapping
    public ResponseEntity<?> crear(@RequestBody Item item){
        if(!categoriaClient.existe(item.getCategoriaId())){
            return ResponseEntity.badRequest()
                    .body("La categoria " + item.getCategoriaId() + " no existe");
        }
        Item guardado = repository.save(item);
        CategoriaDTO categoria = categoriaClient.buscarPorId(guardado.getCategoriaId());
        return ResponseEntity.status(HttpStatus.CREATED).body(ItemResponse.de(guardado, categoria));
    }

    @GetMapping
    public List<Item> listar(){
        return repository.findAll();
    }


    @DeleteMapping("/{id}")
    public ResponseEntity<Void> eliminar(@PathVariable Long id){
        if(!repository.existsById(id)){
            return ResponseEntity.notFound().build();
        }
        repository.deleteById(id);
        return ResponseEntity.noContent().build();
    }

    @GetMapping("/{id}")
    public ResponseEntity<ItemResponse> buscarPorId(@PathVariable Long id){
        return repository.findById(id)
                .map(item -> ResponseEntity.ok(ItemResponse.de(item, categoriaClient.buscarPorId(item.getCategoriaId()))))
                .orElse(ResponseEntity.notFound().build());
    }




}

package com.polec.msusuario.controller;

import com.polec.msusuario.client.RolClient;
import com.polec.msusuario.client.RolDTO;
import com.polec.msusuario.dto.UsuarioResponse;
import com.polec.msusuario.entity.Usuario;
import com.polec.msusuario.repository.UsuarioRepository;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.security.crypto.bcrypt.BCryptPasswordEncoder;
import org.springframework.web.bind.annotation.*;

import java.util.List;

@RestController
@RequestMapping("/usuarios")
public class UsuarioController {
    //inyectar el repository
    private final UsuarioRepository repository;
    //inyectar el objeto rest
    private final RolClient rolClient;
    //crear un objeto para hashear mis password
    private final BCryptPasswordEncoder encoder = new BCryptPasswordEncoder();

    public UsuarioController(UsuarioRepository repository, RolClient rolClient){
        this.repository = repository;
        this.rolClient = rolClient;
    }

    @PostMapping
    public ResponseEntity<?> crear(@RequestBody Usuario usuario){
        if(!rolClient.existe(usuario.getRolId())){
            return ResponseEntity.badRequest()
                    .body("El rol " + usuario.getRolId() + " no existe");
        }
        usuario.setPassword(encoder.encode(usuario.getPassword()));
        Usuario guardado = repository.save(usuario);
        RolDTO rol = rolClient.buscarPorId(guardado.getRolId());
        return ResponseEntity.status(HttpStatus.CREATED).body(UsuarioResponse.de(guardado, rol));
    }

    @GetMapping
    public List<Usuario> listar(){
        return repository.findAll();
    }


    @DeleteMapping("/{id}")
    public ResponseEntity<Void> eliminar(@PathVariable Long id){
        if(!repository.existsById(id)){
            return ResponseEntity.notFound().build();
        }
        repository.deleteById(id);
        return ResponseEntity.noContent().build();
    }

    @GetMapping("/{id}")
    public ResponseEntity<UsuarioResponse> buscarPorId(@PathVariable Long id){
        return repository.findById(id)
                .map(usuario -> ResponseEntity.ok(UsuarioResponse.de(usuario, rolClient.buscarPorId(usuario.getRolId()))))
                .orElse(ResponseEntity.notFound().build());
    }




}
